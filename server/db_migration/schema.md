-- 2. Create the datasets table within that schema
CREATE TABLE datasets (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    month_year TEXT UNIQUE NOT NULL,
    url TEXT NOT NULL,
    row_count BIGINT,
    avg_distance DECIMAL,
    avg_duration DECIMAL,
    outlier_percentage DECIMAL,
    rmse_score DECIMAL,
    added_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    metadata JSONB 
);


CREATE TABLE training_jobs (
    job_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    run_id TEXT UNIQUE, -- The ID generated by MLflow
    model_type TEXT NOT NULL, -- e.g., 'xgboost', 'random_forest'
    parameters JSONB, -- The modified JSON parameters
    status TEXT DEFAULT 'pending', -- 'pending', 'running', 'completed', 'failed'
    val_rmse DECIMAL,
    duration TEXT, -- e.g., '4m 32s'
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    artifact_uri TEXT,
    dataset_ids UUID[] NOT NULL DEFAULT '{}',
    is_registered BOOLEAN DEFAULT FALSE;
);

CREATE TABLE model_registry (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    model_name TEXT NOT NULL, -- e.g., 'trip_duration_predictor'
    version INTEGER NOT NULL, -- Auto-incrementing version per model_name
    run_id TEXT NOT NULL,     -- Foreign Key/Reference to MLflow run_id
    job_id UUID REFERENCES public.training_jobs(job_id),
    status TEXT DEFAULT 'candidate', -- 'candidate', 'staging', 'production', 'archived'
    metrics JSONB,            -- Snapshot of metrics at registration (train_rmse, val_rmse)
    created_at TIMESTAMPTZ DEFAULT NOW(),
    created_by TEXT,
    description TEXT,
    UNIQUE(model_name, version)
);
